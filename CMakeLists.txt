cmake_minimum_required(VERSION 3.16)
project(frost_ecy_stat LANGUAGES CXX)
enable_testing()
add_library(frost components/frost/src/ecy_stat_codec.cpp components/frost/src/ecy_stat_controller.cpp)
target_include_directories(frost PUBLIC components/frost/include)
target_compile_features(frost PUBLIC cxx_std_17)
target_compile_options(frost PRIVATE -Wall -Wextra -Werror)
add_executable(controller_test tests/controller_test.cpp)
target_link_libraries(controller_test PRIVATE frost)
add_executable(schedule_test tests/schedule_test.cpp examples/scheduler/main/schedule.cpp examples/scheduler/main/clock.cpp)
target_include_directories(schedule_test PRIVATE examples/scheduler/main)
target_link_libraries(schedule_test PRIVATE frost)
foreach(test controller_test schedule_test)
  target_compile_options(${test} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME ${test} COMMAND ${test})
endforeach()

add_executable(sntp_test tests/esp_sntp_backend_test.cpp examples/scheduler/main/esp_sntp_backend.cpp examples/scheduler/main/clock.cpp)
target_include_directories(sntp_test PRIVATE tests/esp_stubs examples/scheduler/main components/frost/include)
target_compile_features(sntp_test PRIVATE cxx_std_17)
target_compile_options(sntp_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME sntp_test COMMAND sntp_test)
